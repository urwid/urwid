258ATTR_RE = re.compile(r"(?P<whitespace>[ \n]*)(?P<entry>(?:#[0-9A-Fa-f]{6})|(?:#[0-9A-Fa-f]{3})|(?:[^ \n]+))")
259LONG_ATTR = 7
260SHORT_ATTR = 4  # length of short high-colour descriptions which may
261# be packed one after the next
262
263
264def parse_chart(
265    chart: str,
266    convert: Callable[[str], tuple[urwid.AttrSpec, str] | None],
267) -> list[str | tuple[urwid.AttrSpec, str]]:
268    """Convert string chart into text markup with the correct attributes.
269
270    :param chart: palette chart as a string
271    :param convert: function that converts a single palette entry to an (attr, text) tuple, or ``None`` if no
272                     match is found
273    :returns: text markup for the chart
274    """
275    out = []
276    for match in ATTR_RE.finditer(chart):
277        if match.group("whitespace"):
278            out.append(match.group("whitespace"))
279        entry = match.group("entry")
280        entry = entry.replace("_", " ")
281        while entry:
282            if chart == CHART_TRUE and len(entry) == LONG_ATTR:
283                attrtext = convert(entry[:LONG_ATTR])
284                elen = LONG_ATTR
285            else:
286                elen = SHORT_ATTR
287                attrtext = convert(entry[:SHORT_ATTR])
288            # try the first four characters
289            if attrtext:
290                entry = entry[elen:].strip()
291            else:  # try the whole thing
292                attrtext = convert(entry.strip())
293                assert attrtext, f"Invalid palette entry: {entry!r}"  # noqa: S101  # "assert" is OK for examples
294                elen = len(entry)
295                entry = ""
296            attr, text = attrtext
297            if chart == CHART_TRUE:
298                out.append((attr, "▄"))
299            else:
300                out.append((attr, text.ljust(elen)))
301    return out

304def foreground_chart(
305    chart: str,
306    background: str,
307    colors: Literal[1, 16, 88, 256, 16777216],
308) -> list[str | tuple[urwid.AttrSpec, str]]:
309    """Create text markup for a foreground colour chart.
310
311    :param chart: palette chart as string
312    :param background: colour to use for background of chart
313    :param colors: number of colors
314    :returns: text markup for the chart
315    """
316
317    def convert_foreground(entry: str) -> tuple[urwid.AttrSpec, str] | None:
318        try:
319            attr = urwid.AttrSpec(entry, background, colors)
320        except urwid.AttrSpecError:
321            return None
322        return attr, entry
323
324    return parse_chart(chart, convert_foreground)

327def background_chart(
328    chart: str,
329    foreground: str,
330    colors: Literal[1, 16, 88, 256, 16777216],
331) -> list[str | tuple[urwid.AttrSpec, str]]:
332    """Create text markup for a background colour chart.
333
334    Remaps 8 <= colour < 16 to high-colour versions in the hopes of greater compatibility.
335
336    :param chart: palette chart as string
337    :param foreground: colour to use for foreground of chart
338    :param colors: number of colors
339    :returns: text markup for the chart
340    """
341
342    def convert_background(entry: str) -> tuple[urwid.AttrSpec, str] | None:
343        try:
344            attr = urwid.AttrSpec(foreground, entry, colors)
345        except urwid.AttrSpecError:
346            return None
347        # fix 8 <= colour < 16
348        if colors > 16 and attr.background_basic and attr.background_number >= 8:
349            # use high-colour with same number
350            entry = f"h{attr.background_number:d}"
351            attr = urwid.AttrSpec(foreground, entry, colors)
352        return attr, entry
353
354    return parse_chart(chart, convert_background)

357def main() -> None:
358    """Run the palette test example program."""
359    palette = [
360        ("header", "black,underline", "light gray", "standout,underline", "black,underline", "#88a"),
361        ("panel", "light gray", "dark blue", "", "#ffd", "#00a"),
362        ("focus", "light gray", "dark cyan", "standout", "#ff8", "#806"),
363    ]
364
365    screen = urwid.display.raw.Screen()
366    screen.register_palette(palette)
367
368    lb: urwid.SimpleListWalker[urwid.widget.AbstractFlowWidget] = urwid.SimpleListWalker([])
369    chart_offset = 0  # offset of chart in lb list
370
371    mode_radio_buttons: list[urwid.RadioButton] = []
372    chart_radio_buttons: list[urwid.RadioButton] = []
373
374    def fcs(widget: _Widget) -> urwid.AttrMap[_Widget]:
375        # wrap widgets that can take focus
376        return urwid.AttrMap(widget, None, "focus")
377
378    def set_mode(colors: Literal[1, 16, 88, 256, 16777216], is_foreground_chart: bool) -> None:
379        # set terminal mode and redraw chart
380        screen.set_terminal_properties(colors)
381        screen.reset_default_terminal_palette()
382
383        chart_fn = (background_chart, foreground_chart)[is_foreground_chart]
384        if colors == 1:
385            lb[chart_offset] = urwid.Divider()
386        else:
387            chart: str = {16: CHART_16, 88: CHART_88, 256: CHART_256, 2**24: CHART_TRUE}[colors]
388            txt = chart_fn(chart, "default", colors)
389            lb[chart_offset] = urwid.Text(txt, wrap=urwid.CLIP)
390
391    def on_mode_change(colors: Literal[1, 16, 88, 256, 16777216], rb: urwid.RadioButton, state: bool) -> None:
392        # if this radio button is checked
393        if state:
394            is_foreground_chart = chart_radio_buttons[0].state
395            set_mode(colors, typing.cast("bool", is_foreground_chart))
396
397    def mode_rb(text: str, colors: int, state: bool = False) -> urwid.AttrMap[urwid.RadioButton]:
398        # mode radio buttons
399        rb = urwid.RadioButton(mode_radio_buttons, text, state)
400        urwid.connect_signal(rb, "change", on_mode_change, user_args=(colors,))
401        return fcs(rb)
402
403    def on_chart_change(rb: urwid.RadioButton, state: bool) -> None:
404        # handle foreground check box state change
405        set_mode(screen.colors, state)
406
407    def click_exit(button: urwid.Button) -> typing.NoReturn:
408        raise urwid.ExitMainLoop
409
410    # select_navigation lets TAB / SHIFT TAB move through the mode buttons, then the chart buttons and Exit
411    lb.extend(
412        [
413            urwid.AttrMap(urwid.Text("Urwid Palette Test"), "header"),
414            urwid.AttrMap(
415                urwid.Columns(
416                    [
417                        urwid.Pile(
418                            [
419                                mode_rb("Monochrome", 1),
420                                mode_rb("16-Color", 16, True),
421                                mode_rb("88-Color", 88),
422                                mode_rb("256-Color", 256),
423                                mode_rb("24-bit Color", 2**24),
424                            ],
425                            select_navigation=True,
426                        ),
427                        urwid.Pile(
428                            [
429                                fcs(urwid.RadioButton(chart_radio_buttons, "Foreground Colors", True, on_chart_change)),
430                                fcs(urwid.RadioButton(chart_radio_buttons, "Background Colors")),
431                                urwid.Divider(),
432                                fcs(urwid.Button("Exit", click_exit)),
433                            ],
434                            select_navigation=True,
435                        ),
436                    ],
437                    select_navigation=True,
438                ),
439                "panel",
440            ),
441        ]
442    )
443
444    chart_offset = len(lb)
445    lb.extend([urwid.Divider()])  # placeholder for the chart
446
447    set_mode(16, True)  # displays the chart
448
449    def unhandled_input(key: str | tuple[str, int, int, int]) -> None:
450        if key in {"Q", "q", "esc"}:
451            raise urwid.ExitMainLoop
452
453    urwid.MainLoop(urwid.ListBox(lb, select_navigation=True), screen=screen, unhandled_input=unhandled_input).run()
454

457    main()
