620def check_port(text: str) -> urwid.ValidationMessage | str | None:
621    """Return what is wrong with *text* as a TCP port number, or ``None`` for an empty or usable one."""
622    if not text:
623        return None
624    if not text.isdecimal() or int(text) not in PORTS:
625        return "not a port number"
626    if int(text) in PRIVILEGED_PORTS:
627        return urwid.ValidationMessage(urwid.Severity.WARNING, "a privileged port")
628    return None

631def complete_colour(text: str, pos: int) -> list[tuple[str, int]]:
632    """Return the colours that start with the text before the cursor, none for an empty field."""
633    prefix = text[:pos]
634    return [(name, len(name)) for name in COLOURS if name.startswith(prefix)] if prefix else []
